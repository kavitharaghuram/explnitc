type
    node {
        int data;
        node next;
    }
endtype
decl
    node head;
    node prepend(node h, int v);
    int printList(node h);
    int main();
enddecl

node prepend(node h, int v){
    decl
        node n;
    enddecl
    begin
        n = alloc();
        n.data = v;
        n.next = h;
        return n;
    end
}

int printList(node h){
    decl
        node p;
    enddecl
    begin
        p = h;
        while (p != NULL) do
            write(p.data);
            p = p.next;
        endwhile;
        return 0;
    end
}

int main(){
    decl
        int retval, i;
        node t;
    enddecl
    begin
        retval = initialize();
        head = NULL;
        i = 1;
        while (i <= 3) do
            head = prepend(head, i * 10);
            i = i + 1;
        endwhile;
        retval = printList(head);
        while (head != NULL) do
            t = head.next;
            retval = free(head);
            head = t;
        endwhile;
        write(retval);
        return 0;
    end
}
