decl
    int factorial(int n);
    int main();
enddecl
int factorial(int n){
    decl
        int f;
    enddecl
    begin
        if (n == 1) then
            f = 1;
        else
            f = n * factorial(n - 1);
        endif;
        return f;
    end
}
int main(){
    decl
        int x;
    enddecl
    begin
        x = 3;
        write(factorial(x));
        return 0;
    end
}
