decl
    tuple student(int roll, int year) a, b, *p;
    int show(tuple student s);
    int bump(tuple student *q);
    tuple student make(int r, int y);
    int main();
enddecl

int show(tuple student s){
    begin
        write(s.roll);
        write(s.year);
        return 0;
    end
}
int bump(tuple student *q){
    begin
        q.roll = q.roll + 1;
        return 0;
    end
}
tuple student make(int r, int y){
    decl
        tuple student t;
    enddecl
    begin
        t.roll = r;
        t.year = y;
        return t;
    end
}
int main(){
    decl
        int d;
    enddecl
    begin
        a = make(5, 2020);
        show(a);
        p = &a;
        bump(p);
        b = a;
        show(b);
        return 0;
    end
}
