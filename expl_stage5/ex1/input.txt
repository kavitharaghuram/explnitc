decl
    int swap(int *p, int *q);
    int main();
enddecl
int swap(int *p, int *q){
    decl
        int t;
    enddecl
    begin
        t = *p;
        *p = *q;
        *q = t;
        return 0;
    end
}
int main(){
    decl
        int a, b;
    enddecl
    begin
        a = 5;
        b = 9;
        swap(&a, &b);
        write(a);
        write(b);
        return 0;
    end
}
